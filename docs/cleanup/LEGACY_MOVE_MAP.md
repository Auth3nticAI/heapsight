## Restoration Script

To restore all quarantined files to their original locations:

```javascript
// Run with: node restore.js  (from repo root)
const fs = require('fs');
const path = require('path');
const root = __dirname;

const files = Array.from({length: 50}, (_, i) => {
  const n = String(i + 1).padStart(2, '0');
  return `src/data/game-templates/simple-rpg/lesson-${n}-rpg.ts`;
});

files.forEach(f => {
  const src = path.join(root, 'legacy/moved', f);
  const dst = path.join(root, f);
  fs.copyFileSync(src, dst);
  console.log('restored:', f);
});
```
